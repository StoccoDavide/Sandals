256 bool single_shooting(VectorX const & t_mesh, VectorF const & ics)
257 {
258 #define CMD "Sandals::BoundaryValueProblem::single_shooting(...): "
259
260 using ShootingF = Eigen::Matrix<Real, 2*N, 1>;
261 using ShootingJF = Eigen::Matrix<Real, 2*N, 2*N>;
262
263 // Temporary variables
264 VectorF b, x_ini, x_end;
265 ShootingF x_sol, x_step, b_sys;
266 ShootingJF A_sys;
267 const Integer local_intervals{std::max(1, this->m_subintervals)};
268 const Integer num_intervals{static_cast<Integer>(t_mesh.size()) - 1};
269 Integer idx_x_ini{0}, idx_x_end{num_intervals};
270 if (this->m_integrator->reverse_mode()) {idx_x_ini = num_intervals; idx_x_end = 0;}
271
272 SANDALS_ASSERT_WARNING(local_intervals == 1,
273 CMD "please avoid using multiple subintervals with single shooting method, just resize the "
274 "mesh accordingly.");
275
276 // Prepare the linear system for the Newton step
277 A_sys.template block<N, N>(0, N).setIdentity();
278
279 // Initialize the guess and the solution
280 x_sol << ics, ics;
281 this->m_solution->clear();
282
283 // Solve the boundary value problem using a linearized Newton method
284 MatrixJX Jx(MatrixJX::Identity());
285 Eigen::ColPivHouseholderQR<ShootingJF> qr;
286 for (Integer iter{0}; iter < this->m_max_iterations; ++iter) {
287
288 /* Single shooting method scheme
289 [A_sys] {x} = {b_sys}
290 / -Jx I \ / \ / x_end - x_sol_end \
291 | | | dx | = | |
292 \ Jb_x_ini Jb_x_end / \ / \ -b /
293 */
294
295 // Integrate the system and propagate the Jacobian
296 if (!this->m_integrator->solve(t_mesh, x_sol.template head<N>(), *this->m_solution, Jx)) {
297 SANDALS_ERROR(CMD "failed to integrate the system " << this->m_integrator->name() << ".");
298 return false;
299 }
300
301 // Retrieve the initial and final states
302 x_ini = this->m_solution->x.col(idx_x_ini);
303 x_end = this->m_solution->x.col(idx_x_end);
304
305 // Evaluate the residual of the boundary conditions
306 b = this->b(x_ini, x_end);
307
308 // Print the iteration info
309 if (this->m_verbose) {
310 std::cout
311 << "Iteration " << iter << ": |b| = " << b.norm() << std::endl
312 << " x(" << t_mesh(idx_x_ini) << ") = " << x_ini.transpose() << std::endl
313 << " x(" << t_mesh(idx_x_end) << ") = " << x_end.transpose() << std::endl;
314 }
315
316 // Check convergence
317 if (b.norm() < this->m_tolerance) {return true;}
318
319 // Build the linear system
320 b_sys.template head<N>() = x_end - x_sol.template tail<N>();
321 b_sys.template tail<N>() = -b;
322 A_sys.template block<N, N>(0, 0) = -Jx;
323 if (this->m_integrator->reverse_mode()) {
324 A_sys.template block<N, N>(N, N) = this->Jb_x_ini(x_ini, x_end);
325 A_sys.template block<N, N>(N, 0) = this->Jb_x_end(x_ini, x_end);
326 } else {
327 A_sys.template block<N, N>(N, 0) = this->Jb_x_ini(x_ini, x_end);
328 A_sys.template block<N, N>(N, N) = this->Jb_x_end(x_ini, x_end);
329 }
330
331 // Compute the solution of the linear system
332 qr.compute(A_sys);
333 SANDALS_ASSERT(qr.rank() == qr.cols(), CMD "singular Jacobian detected.");
334 x_sol += qr.solve(b_sys);
335 }
336
337 // If the loop completes without returning, indicate failure
338 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
339 return false;
340
341 #undef CMD
342 }

351 bool multiple_shooting(VectorX const & t_mesh, MatrixX const & x_guess)
352 {
353 #define CMD "Sandals::BoundaryValueProblem::multiple_shooting(...): "
354
355 using DynVec = Eigen::Matrix<Real, Eigen::Dynamic, 1>;
356 using DynMat = Eigen::Matrix<Real, Eigen::Dynamic, Eigen::Dynamic>;
357
358 // Check if the mesh and the guess have compatible sizes
359 SANDALS_ASSERT(t_mesh.size() > 1,
360 CMD "expected at least two time mesh points.");
361 SANDALS_ASSERT(x_guess.cols() == t_mesh.size(),
362 CMD "incompatible sizes between time mesh and states guess.");
363
364 // Temporary variables
365 VectorF x_ini, x_end;
366 const Integer num_intervals{static_cast<Integer>(t_mesh.size()) - 1};
367 const Integer local_intervals{std::max(1, this->m_subintervals)};
368 const Integer c_size{num_intervals*N};
369 const Integer x_size{(num_intervals + 1)*N};
370 Integer idx_x_ini{0}, idx_x_end{num_intervals};
371 if (this->m_integrator->reverse_mode()) {idx_x_ini = num_intervals; idx_x_end = 0;}
372
373 // Prepare the linear system for the Newton step
374 DynVec b_sys(c_size + N);
375 DynMat A_sys(DynMat::Zero(x_size, x_size));
376 for (Integer k{0}; k < num_intervals; ++k) {
377 A_sys.template block<N, N>(k*N, (k + 1)*N).setIdentity();
378 }
379
380 // Initialize the solution
381 this->m_solution->clear();
382 this->m_solution->resize(t_mesh.size());
383 this->m_solution->t = t_mesh;
384 this->m_solution->h.setZero();
385
386 // Initial guess for the states as the guess at the mesh nodes
387 MatrixX x_sol(x_guess);
388
389 // Solve the boundary value problem using a linearized Newton method
390 VectorX t_local_mesh;
391 Solution<Real, N, M> local_sol;
392 MatrixJX Jx;
393 Eigen::ColPivHouseholderQR<DynMat> qr;
394 for (Integer iter{0}; iter < this->m_max_iterations; ++iter) {
395
396 /* Multiple shooting method scheme
397 [A_sys] {x} = {b_sys}
398 / -Jx_1 I \ / : \ / x_ini_2 - x_sol_1 \
399 | . . | | : | | : |
400 | . . | | dx | = | : |
401 | -Jx_n I | | : | | x_ini_n+1 - x_sol_n |
402 \ Jb_x_ini Jb_x_end / \ : / \ -b /
403 */
404
405 // Integrate each interval with the local mesh
406 for (Integer k{0}; k < num_intervals; ++k) {
407 t_local_mesh = VectorX::LinSpaced(local_intervals+1, t_mesh(k), t_mesh(k + 1));
408 Jx.setIdentity();
409 if (!this->m_integrator->template solve<true>(t_local_mesh, x_sol.col(k), local_sol, Jx)) {
410 SANDALS_ERROR(CMD "failed to integrate interval " << k << ".");
411 return false;
412 }
413 // Store the local solution
414 if (k == 0) {
415 this->m_solution->x.col(0) = local_sol.x.col(0);
416 this->m_solution->h.col(0) = local_sol.h.col(0);
417 }
418 this->m_solution->x.col(k + 1) = local_sol.x.col(local_intervals);
419 this->m_solution->h.col(k + 1) = local_sol.h.col(local_intervals);
420
421 // Jacobian propagation for the current interval
422 A_sys.template block<N, N>(k*N, k*N) = -Jx;
423
424 // Continuity residual for interior nodes
425 b_sys.template segment<N>(k*N) = this->m_solution->x.col(k + 1) - x_sol.col(k + 1);
426 }
427
428 // Update the boundary condition states
429 x_ini = this->m_solution->x.col(idx_x_ini);
430 x_end = this->m_solution->x.col(idx_x_end);
431
432 // Boundary condition residuals
433 b_sys.template tail<N>() = -this->b(x_ini, x_end);
434
435 // Print the iteration info
436 if (this->m_verbose) {
437 std::cout
438 << "Iteration " << iter << ": |b| = " << b_sys.norm() << std::endl
439 << " x(" << t_mesh(idx_x_ini) << ") = " << x_ini.transpose() << std::endl
440 << " x(" << t_mesh(idx_x_end) << ") = " << x_end.transpose() << std::endl;
441 }
442
443 // Check convergence
444 if (b_sys.norm() < this->m_tolerance) {return true;}
445
446 // Update the boundary condition Jacobian blocks
447 A_sys.template block<N, N>(c_size, idx_x_ini*N) = this->Jb_x_ini(x_ini, x_end);
448 A_sys.template block<N, N>(c_size, idx_x_end*N) = this->Jb_x_end(x_ini, x_end);
449
450 // Solve the linear system
451 qr.compute(A_sys);
452 SANDALS_ASSERT(qr.rank() == qr.cols(), CMD "singular linear system detected.");
453 x_sol += qr.solve(b_sys).reshaped(N, num_intervals + 1);
454 }
455
456 // If the loop completes without returning, indicate failure
457 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
458 return false;
459
460 #undef CMD
461 }
