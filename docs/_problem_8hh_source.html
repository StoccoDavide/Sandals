225 bool single_shooting(VectorX const & t_mesh, VectorF const & ics)
226 {
227 using ShootingF = Eigen::Matrix<Real, 2*N, 1>;
228 using ShootingJF = Eigen::Matrix<Real, 2*N, 2*N>;
229
230 #define CMD "Sandals::Problem::single_shooting(...): "
231
232 // Temporary variables
233 VectorF b, x_ini, x_end;
234 ShootingF x_sol, x_step, b_sys;
235 ShootingJF A_sys;
236 A_sys.template block<N, N>(0, N) = MatrixJX::Identity();
237
238 // Initialize the guess and the solution
239 x_sol << ics, ics;
240 this->m_solution->clear();
241
242 // Solve the boundary value problem using a linearized Newton method
243 MatrixJX Jx(MatrixJX::Identity());
244 Eigen::FullPivLU<ShootingJF> lu;
245 for (Integer iter{0}; iter < this->m_max_iterations; ++iter) {
246
247 /* Single shooting method
248 [A_sys] {x} = {b_sys}
249 / -Jx I \ / \ / x_end - x_sol_end \
250 | | | dx | = | |
251 \ Jb_x_ini Jb_x_end / \ / \ -b /
252 */
253
254 // Integrate the system and propagate the Jacobian
255 if (!this->m_integrator->solve(t_mesh, x_sol.template head<N>(), *this->m_solution, Jx)) {
256 SANDALS_ERROR(CMD "failed to integrate the system " << this->m_integrator->name() << ".");
257 return false;
258 }
259
260 // Retrieve the initial and final states
261 x_ini = x_sol.template head<N>();
262 x_end = this->m_solution->x.col(this->m_solution->t.size() - 1);
263
264 // Evaluate the residual of the boundary conditions
265 b = this->b(x_ini, x_end);
266
267 // Print the iteration info
268 if (this->m_verbose) {
269 std::cout
270 << "Iteration " << iter << ": |b| = " << b.norm() << std::endl
271 << " x(" << t_mesh.template head<1>() << ") = " << x_ini.transpose() << std::endl
272 << " x(" << t_mesh.template tail<1>() << ") = " << x_end.transpose() << std::endl;
273 }
274
275 // Check if the solution is found (i.e., if the boundary conditions are satisfied)
276 if (b.norm() < this->m_tolerance) {return true;}
277
278 // Build the linear system
279 b_sys.template head<N>() = x_end - x_sol.template tail<N>();
280 b_sys.template tail<N>() = -b;
281 A_sys.template block<N, N>(0, 0) = -Jx;
282 A_sys.template block<N, N>(N, 0) = this->Jb_x_ini(x_ini, x_end);
283 A_sys.template block<N, N>(N, N) = this->Jb_x_end(x_ini, x_end);
284
285 // Compute the solution of the linear system
286 lu.compute(A_sys);
287 SANDALS_ASSERT(lu.rank() == 2*N, CMD "singular Jacobian detected.");
288 x_step = lu.solve(b_sys);
289
290 // Update the solution
291 x_sol += x_step;
292 }
293
294 // If the loop completes without returning, indicate failure
295 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
296 return false;
297
298 #undef CMD
299 }

307 bool multiple_shooting(VectorX const & t_mesh, VectorF const & ics)
308 {
309 using DynVec = Eigen::Matrix<Real, Eigen::Dynamic, 1>;
310 using DynMat = Eigen::Matrix<Real, Eigen::Dynamic, Eigen::Dynamic>;
311
312 #define CMD "Sandals::Problem::multiple_shooting(...): "
313
314 // Temporary variables
315 const Integer num_intervals{static_cast<Integer>(t_mesh.size()) - 1};
316
317 // Initialize the guess and the solution
318 std::vector<VectorF> x_guess(num_intervals + 1, ics);
319 this->m_solution->clear();
320 this->m_solution->resize(static_cast<Integer>(t_mesh.size()));
321
322 // Solve the boundary value problem using a linearized Newton method
323 std::vector<MatrixJX> Jx(num_intervals, MatrixJX::Identity());
324 for (Integer iter{0}; iter < this->m_max_iterations; ++iter) {
325
326 // Store states at interval boundaries
327 std::vector<VectorF> x_states(num_intervals + 1);
328 x_states[0] = x_guess[0];
329
330 // Integrate each interval
331 Solution<Real, N, M> local_sol;
332 for (Integer k{0}; k < num_intervals; ++k) {
333 // The integrator expects the two-node segment [t_k, t_{k+1}]
334 Integer local_intervals{std::max(1, this->m_subintervals)};
335 VectorX t_local_mesh(VectorX::LinSpaced(local_intervals+1, t_mesh(k), t_mesh(k + 1)));
336 if (!this->m_integrator->solve(t_local_mesh, x_guess[k], local_sol, Jx[k])) {
337 SANDALS_ERROR(CMD "failed to integrate interval " << k << ".");
338 return false;
339 }
340 // Save last state of the local solution into the states array
341 x_states[k + 1] = local_sol.x.col(local_intervals);
342 // Save the local solution into the global solution (avoid duplicating boundary points)
343 if (k == 0) {
344 this->m_solution->t(k) = local_sol.t(0);
345 this->m_solution->t(k + 1) = local_sol.t(local_intervals);
346 this->m_solution->x.col(k) = local_sol.x.col(0);
347 this->m_solution->x.col(k + 1) = local_sol.x.col(local_intervals);
348 } else {
349 this->m_solution->t(k + 1) = local_sol.t(local_intervals);
350 this->m_solution->x.col(k + 1) = local_sol.x.col(local_intervals);
351 }
352 }
353
354 // Build continuity residual for states at interior nodes
355 const Integer c_size{num_intervals*N};
356 const Integer x_size{(num_intervals + 1)*N}; // state unknowns at every mesh node
357 DynVec residual(DynVec::Zero(c_size));
358 for (Integer k{0}; k < num_intervals; ++k) {
359 residual.segment(k*N, N) = x_states[k + 1] - x_guess[k + 1];
360 }
361
362 // Boundary condition residuals
363 VectorF b(this->b(x_states[0], x_states[num_intervals]));
364 const Integer b_size{static_cast<Integer>(b.size())};
365
366 // Check size consistency
367 SANDALS_ASSERT(c_size + b_size == x_size,
368 CMD "expected square linear system, got " << (c_size + b_size) << " x " << x_size << ".");
369
370 // Total residual vector: [continuity; -b]
371 DynVec b_sys(DynVec::Zero(c_size + b_size));
372 b_sys.head(c_size) = residual;
373 b_sys.segment(c_size, b_size) = -b;
374
375 // Print the iteration info
376 if (this->m_verbose) {
377 std::cout
378 << "Iteration " << iter << ": |b| = " << b.norm() << ", |c| = " << residual.norm() << std::endl
379 << " x(" << t_mesh.template head<1>() << ") = " << x_states[0].transpose() << std::endl
380 << " x(" << t_mesh.template tail<1>() << ") = " << x_states[num_intervals].transpose() << std::endl;
381 }
382
383 // Check convergence
384 if (b.norm() < this->m_tolerance && residual.norm() < this->m_tolerance) {return true;}
385
386 // Build Jacobian matrix for Newton step
387 DynMat A_sys(DynMat::Zero(x_size, x_size));
388 for (Integer k{0}; k < num_intervals; ++k) {
389 A_sys.block(k*N, k*N, N, N) = -MatrixJX::Identity(); // -I at block (k, k)
390 A_sys.block(k*N, (k + 1)*N, N, N) = MatrixJX::Identity(); // +I at block (k, k+1)
391 }
392 MatrixJF Jb_ini(this->Jb_x_ini(x_states[0], x_states[num_intervals]));
393 MatrixJF Jb_end(this->Jb_x_end(x_states[0], x_states[num_intervals]));
394 A_sys.block(c_size, 0, Jb_ini.rows(), N) = Jb_ini;
395 A_sys.block(c_size, num_intervals*N, Jb_end.rows(), N) = Jb_end;
396
397 // Solve the linear system
398 Eigen::FullPivLU<DynMat> lu(A_sys);
399 if (lu.rank() < lu.cols()) {
400 SANDALS_ERROR(CMD "singular linear system detected.");
401 return false;
402 }
403 DynVec x_step(lu.solve(b_sys));
404
405 // Update guesses
406 for (Integer k{0}; k < num_intervals + 1; ++k) {
407 x_guess[k] += x_step.template segment<N>(k*N);
408 }
409 }
410
411 // If the loop completes without returning, indicate failure
412 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
413 return false;
414
415 #undef CMD
416 }
