641 Real estimate_step(VectorN const & x, VectorN const & x_e, Real const h_k) const
642 {
643 Real desired_error{this->m_absolute_tolerance + this->m_relative_tolerance *
644 std::max(x.array().abs().maxCoeff(), x_e.array().abs().maxCoeff())};
645 Real truncation_error{(x - x_e).array().abs().maxCoeff()};
646 return h_k * std::min(this->m_max_safety_factor, std::max(this->m_min_safety_factor,
647 this->m_safety_factor * std::pow(desired_error/truncation_error,
648 1.0/std::max(this->m_tableau.order, this->m_tableau.order_e))));
649 }

708 bool erk_explicit_step(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
709 Real & h_new, MatrixK & K) const
710 {
711 using Eigen::all;
712 using Eigen::seqN;
713
714 // Compute the K variables in the case of an explicit method and explicit system
715 VectorN x_node;
716 for (Integer i{0}; i < S; ++i) {
717 x_node = x_old + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
718 if (!this->m_reverse) {
719 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f(x_node, t_old + h_old*this->m_tableau.c(i));
720 } else {
721 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f_reverse(x_node, t_old + h_old*this->m_tableau.c(i));
722 }
723 }
724 if (!K.allFinite()) {return false;}
725
726 // Perform the step and obtain the next state
727 x_new = x_old + K * this->m_tableau.b;
728
729 // Adapt next step
730 if (this->m_adaptive && this->m_tableau.is_embedded) {
731 VectorN x_emb = x_old + K * this->m_tableau.b_e;
732 h_new = this->estimate_step(x_new, x_emb, h_old);
733 }
734 return true;
735 }

750 bool erk_explicit_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
751 MatrixJX & Jx) const
752 {
753 using Eigen::all;
754 using Eigen::seqN;
755
756 // Propagate the derivative of K with respect to x
757 VectorN x_node;
758 MatrixN Jf_x;
759 std::array<MatrixN, S> dK_dx;
760 for (Integer i{0}; i < S; ++i) {
761 // Compute the node
762 x_node = x + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
763
764 // Compute the Jacobian of f with respect to x at the node
765 if (!this->m_reverse) {
766 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x(x_node, t + h*this->m_tableau.c(i));
767 } else {
768 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x_reverse(x_node, t + h*this->m_tableau.c(i));
769 }
770
771 // Propagate the derivative of K with respect to x
772 dK_dx[i] = h * Jf_x;
773 for (Integer j{0}; j < i; ++j) {
774 dK_dx[i] += h * Jf_x * dK_dx[j] * this->m_tableau.A(i, j);
775 }
776
777 // Check for NaNs or Infs
778 if (!dK_dx[i].allFinite()) {return false;}
779 }
780
781 // Compute the derivative of x_new with respect to x
782 Jx.setIdentity();
783 for (Integer i{0}; i < S; ++i) {Jx += h * dK_dx[i] * this->m_tableau.b(i);}
784
785 return true;
786 }

879 bool erk_implicit_step(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
880 Real & h_new, MatrixK & K) const
881 {
882 VectorN K_sol;
883 VectorN K_ini(VectorN::Zero());
884
885 // Check if the solver converged
886 for (Integer s{0}; s < S; ++s) {
887 if (this->m_newtonX.solve(
888 [this, s, &K, &x_old, t_old, h_old](VectorN const & K_fun, VectorN & fun)
889 {K.col(s) = K_fun; this->erk_implicit_function(s, x_old, t_old, h_old, K, fun);},
890 [this, s, &K, &x_old, t_old, h_old](VectorN const & K_jac, MatrixN & jac)
891 {K.col(s) = K_jac; this->erk_implicit_jacobian(s, x_old, t_old, h_old, K, jac);},
892 K_ini, K_sol)) {
893 K.col(s) = K_sol;
894 } else {
895 return false;
896 }
897 }
898
899 // Perform the step and obtain the next state
900 x_new = x_old + K * this->m_tableau.b;
901
902 // Adapt next step
903 if (this->m_adaptive && this->m_tableau.is_embedded) {
904 VectorN x_emb(x_old + K * this->m_tableau.b_e);
905 h_new = this->estimate_step(x_new, x_emb, h_old);
906 }
907 return true;
908 }

923 bool erk_implicit_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
924 MatrixJX & Jx) const
925 {
926 using Eigen::all;
927 using Eigen::seqN;
928
929 // Propagate the derivative of K with respect to x
930 VectorN x_node, x_dot_node;
931 MatrixN JF_x, JF_x_dot, A, b;
932 std::array<MatrixN, S> dK_dx;
933 Eigen::FullPivLU<MatrixN> lu;
934 for (Integer i{0}; i < S; ++i) {
935 // Compute the node
936 x_node = x + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
937 x_dot_node = K.col(i) / h;
938
939 // Compute the Aobians of F with respect to x and x_dot at the node
940 if (!this->m_reverse) {
941 JF_x = this->m_system->JF_x(x_node, x_dot_node, t + h*this->m_tableau.c(i));
942 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t + h*this->m_tableau.c(i));
943 } else {
944 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
945 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
946 }
947
948 // Propagate the derivative of K with respect to x
949 A = JF_x_dot;
950 b = -h * JF_x;
951 for (Integer j{0}; j < i; ++j) {
952 b -= h * this->m_tableau.A(i, j) * JF_x * dK_dx[j];
953 }
954
955 // Solve the linear system
956 lu.compute(A);
957 dK_dx[i] = lu.solve(b);
958
959 // Check for NaNs or Infs
960 if (!dK_dx[i].allFinite()) {return false;}
961 }
962
963 // Compute the derivative of x_new with respect to x
964 Jx.setIdentity();
965 for (Integer i{0}; i < S; ++i) {Jx += h * dK_dx[i] * this->m_tableau.b(i);}
966

1055 void irk_jacobian(VectorN const & x, Real const t, Real const h, VectorK const & K, MatrixJ & jac) const
1056 {
1057 using Eigen::seqN;
1058
1059 // Reset the Jacobian matrix
1060 jac.setZero();
1061
1062 // Loop through each equation of the system
1063 MatrixK K_mat{K.reshaped(N, S)};
1064 Real t_node;
1065 VectorN x_node, x_dot_node;
1066 MatrixN JF_x, JF_x_dot;
1067 auto idx = seqN(0, N), jdx = seqN(0, N);
1068 for (Integer i{0}; i < S; ++i) {
1069 t_node = t + h * this->m_tableau.c(i);
1070 x_node = x + K_mat * this->m_tableau.A.row(i).transpose();
1071
1072 // Compute the Jacobians with respect to x and x_dot
1073 x_dot_node = K_mat.col(i) / h;
1074 if (!this->m_reverse) {
1075 JF_x = this->m_system->JF_x(x_node, x_dot_node, t_node);
1076 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t_node);
1077 } else {
1078 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t_node);
1079 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t_node);
1080 }
1081
1082 // Combine the Jacobians with respect to x and x_dot to obtain the Jacobian with respect to K
1083 idx = seqN(i*N, N);
1084 for (Integer j{0}; j < S; ++j) {
1085 jdx = seqN(j*N, N);
1086 if (i == j) {
1087 jac(idx, jdx) = this->m_tableau.A(i, j) * JF_x + JF_x_dot / h;
1088 } else {
1089 jac(idx, jdx) = this->m_tableau.A(i, j) * JF_x;
1090 }
1091 }
1092 }

1152 bool irk_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
1153 MatrixJX & Jx) const
1154 {
1155 using Eigen::all;
1156 using Eigen::seqN;
1157
1158 // Propagate the derivative of K with respect to x
1159 VectorN x_node, x_dot_node;
1160 MatrixN JF_x, JF_x_dot;
1161 Eigen::Matrix<Real, N*S, N*S> A;
1162 Eigen::Matrix<Real, N*S, N> b;
1163 std::array<MatrixN, S> dK_dx;
1164 Eigen::FullPivLU<Eigen::Matrix<Real, N*S, N*S>> lu;
1165 for (Integer i{0}; i < S; ++i) {
1166 // Compute the node
1167 x_node = x + K * this->m_tableau.A.row(i).transpose();
1168 x_dot_node = K.col(i) / h;
1169
1170 // Compute the Jacobians of F with respect to x and x_dot at the node
1171 if (!this->m_reverse) {
1172 JF_x = this->m_system->JF_x(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1173 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1174 } else {
1175 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1176 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1177 }
1178
1179 // Fill the big linear system
1180 A.template block<N, N>(i*N, i*N) = JF_x_dot;
1181 for (Integer j{0}; j < S; ++j) {
1182 A.template block<N, N>(i*N, j*N) += h * this->m_tableau.A(i, j) * JF_x;
1183 }
1184 b.template block<N, N>(i*N, 0) = -h * JF_x;
1185
1186 // Check for NaNs or Infs
1187 if (!JF_x.allFinite() || !JF_x_dot.allFinite()) {return false;}
1188 }
1189
1190 // Solve the big linear system
1191 lu.compute(A);
1192 Eigen::Matrix<Real, N*S, N> dK_dx_mat(lu.solve(b));
1193 if (!dK_dx_mat.allFinite()) {return false;}
1194
1195 // Reshape the solution
1196 for (Integer i{0}; i < S; ++i) {
1197 dK_dx[i] = dK_dx_mat.template block<N, N>(i*N, 0);
1198 }
1199
1200 // Compute the derivative of x_new with respect to x
1201 Jx.setIdentity();
1202 for (Integer i{0}; i < S; ++i) {Jx += h * dK_dx[i] * this->m_tableau.b(i);}
1203
1204 return true;
1205 }

1357 bool dirk_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
1358 MatrixJX & Jx) const
1359 {
1360 using Eigen::all;
1361 using Eigen::seqN;
1362
1363 // Propagate the derivative of K with respect to x for DIRK methods
1364 VectorN x_node, x_dot_node;
1365 MatrixN JF_x, JF_x_dot, A, b;
1366 std::array<MatrixN, S> dK_dx;
1367 Eigen::FullPivLU<MatrixN> lu;
1368 for (Integer i{0}; i < S; ++i) {
1369 // Compute the node
1370 x_node = x + K(all, seqN(0, i+1)) * this->m_tableau.A(i, seqN(0, i+1)).transpose();
1371 x_dot_node = K.col(i) / h;
1372
1373 // Compute the Jacobians of F with respect to x and x_dot at the node
1374 if (!this->m_reverse) {
1375 JF_x = this->m_system->JF_x(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1376 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1377 } else {
1378 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1379 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1380 }
1381
1382 // Propagate the derivative of K with respect to x
1383 A = JF_x_dot;
1384 b = -h * JF_x;
1385 for (Integer j{0}; j < i; ++j) {
1386 b -= h * this->m_tableau.A(i, j) * JF_x * dK_dx[j];
1387 }
1388
1389 // Solve the linear system
1390 lu.compute(A);
1391 dK_dx[i] = lu.solve(b);
1392
1393 // Check for NaNs or Infs
1394 if (!dK_dx[i].allFinite()) {return false;}
1395 }
1396
1397 // Compute the derivative of x_new with respect to x
1398 Jx.setIdentity();
1399 for (Integer i{0}; i < S; ++i) {Jx += h * dK_dx[i] * this->m_tableau.b(i);}
1400
1401 return true;
1402 }

1485 template <bool Propagate = true>
1486 bool advance(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
1487 Real & h_new, MatrixJX & Jx) const
1488 {
1489 #define CMD "Sandals::RungeKutta::advance(...): "
1490
1491 // Check step size
1492 SANDALS_ASSERT(h_old > Real(0.0), CMD "in " << this->m_tableau.name << " solver, h = "<<
1493 h_old << ", expected > 0.");
1494
1495 // Reset the derivative propagation matrix
1496 if constexpr (Propagate) {Jx.setIdentity();}
1497
1498 // If the integration step failed, try again with substepping
1499 MatrixK K;
1500 if (!this->step(x_old, t_old, h_old, x_new, h_new, K)) {
1501
1502 // Store temporary variables
1503 VectorN x_tmp(x_old);
1504 Real t_tmp{t_old}, h_tmp{h_old / Real(2.0)};
1505
1506 // Substepping logic
1507 Integer max_k{this->m_max_substeps * this->m_max_substeps}, k{2};
1508 Real h_new_tmp;
1509 while (k > 0) {
1510 // Calculate the next step with substepping logic
1511 if (this->step(x_tmp, t_tmp, h_tmp, x_new, h_new_tmp, K)) {
1512
1513 if constexpr (Propagate) {
1514 // Propagate the derivative of K with respect to x
1515 MatrixJX Jx_tmp;
1516 if (!this->propagate(x_tmp, t_tmp, h_tmp, K, Jx_tmp)) {
1517 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1518 ", Jacobian propagation failed, aborting.");
1519 return false;
1520 }
1521
1522 // Update the derivative propagation
1523 if constexpr (Propagate) {Jx *= Jx_tmp;}
1524 }
1525
1526 // Accept the step
1527 h_tmp = h_new_tmp;
1528
1529 // If substepping is enabled, double the step size
1530 if (k > 0 && k < max_k) {
1531 k -= 1;
1532 // If the substepping index is even, double the step size
1533 if (k % 2 == 0) {
1534 h_tmp = Real(2.0) * h_tmp;
1535 if (this->m_verbose) {
1536 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1537 ", integration succedded disable one substepping layer.");
1538 }
1539 }
1540 }
1541
1542 // Check the infinity norm of the solution
1543 SANDALS_ASSERT(std::isfinite(x_tmp.maxCoeff()), CMD "in " << this->m_tableau.name <<
1544 " solver, at t = " << t_tmp << ", ||x||_inf = inf, computation interrupted.");
1545
1546 } else {
1547
1548 // If the substepping index is too high, abort the integration
1549 k += 2;
1550 SANDALS_ASSERT(k < max_k, CMD "in " << this->m_tableau.name << " solver, at t = " <<
1551 t_tmp << ", integration failed with h = " << h_tmp << ", aborting.");
1552 return false;
1553
1554 // Otherwise, try again with a smaller step
1555 if (this->m_verbose) {SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, " <<
1556 "at t = " << t_tmp << ", integration failed, adding substepping layer.");}
1557 h_tmp /= Real(2.0);
1558 continue;
1559 }
1560
1561 // Store independent variable (or time)
1562 t_tmp += h_tmp;
1563 }
1564
1565 // Store output states substepping solutions
1566 x_new = x_tmp;
1567 h_new = h_tmp;
1568
1569 } else {
1570 // Propagate the derivative of the solution with respect to the states x
1571 if constexpr (Propagate) {
1572 if (!this->propagate(x_old, t_old, h_old, K, Jx)) {
1573 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_old <<
1574 ", Jacobian propagation failed, aborting.");
1575 return false;
1576 }
1577 }
1578 }
1579
1580 // Project intermediate solution on the invariants
1581 if (this->m_projection) {
1582 VectorN x_projected;
1583 if (this->project(x_new, t_old + h_new, x_projected)) {
1584 x_new = x_projected;
1585 } else {
1586 return false;
1587 }
1588 }
1589 return true;
1590
1591 #undef CMD
1592 }

1608 template <bool Propagate = true>
1609 bool solve(VectorX const & t_mesh, VectorN const & ics, Solution<Real, N, M> & sol, MatrixJX & Jx) const
1610 {
1611 using Eigen::last;
1612
1613 #define CMD "Sandals::RungeKutta::solve(...): "
1614
1615 // Instantiate output
1616 sol.resize(static_cast<Integer>(t_mesh.size()));
1617
1618 // Store initial conditions
1619 sol.t(0) = t_mesh(0);
1620 sol.x.col(0) = ics;
1621 sol.h.col(0) = this->m_system->h(ics, t_mesh(0));
1622
1623 // Check initial conditions
1624 if constexpr (Propagate) {
1625 if (!Jx.allFinite()) {
1626 SANDALS_ERROR(CMD "in " << this->m_tableau.name << " solver, initial Jacobian " <<
1627 "contains NaNs or Infs, aborting.");
1628 return false;
1629 }
1630 }
1631
1632 // Update the current step
1633 Integer step{0};
1634 VectorN x_old_step(ics), x_new_step(ics);
1635 MatrixJX Jx_step;
1636 Real t_step{t_mesh(0)}, h_step{t_mesh(1) - t_mesh(0)}, h_tmp_step{h_step}, h_new_step;
1637 bool mesh_point_bool, saturation_bool;
1638
1639 while (true) {
1640 // Integrate system
1641 if (!this->template advance<Propagate>(x_old_step, t_step, h_step, x_new_step, h_new_step, Jx_step))
1642 {return false;}
1643
1644 // Update the current step
1645 t_step += h_step;
1646
1647 // Saturate the suggested timestep
1648 mesh_point_bool = std::abs(t_step - t_mesh(step+1)) < SQRT_EPSILON;
1649 saturation_bool = t_step + h_new_step > t_mesh(step+1) + SQRT_EPSILON;
1650 if (this->m_adaptive && this->m_tableau.is_embedded && !mesh_point_bool && saturation_bool) {
1651 h_tmp_step = h_new_step; // Used to store the previous step and keep the integration pace
1652 h_step = t_mesh(step+1) - t_step;
1653 } else {
1654 h_step = h_new_step;
1655 }
1656
1657 // Store solution if the step is a mesh point
1658 if (!this->m_adaptive || mesh_point_bool) {
1659
1660 // Update temporaries
1661 step += 1;
1662 h_step = h_tmp_step;
1663
1664 // Update outputs
1665 sol.t(step) = t_step;
1666 sol.x.col(step) = x_new_step;
1667 sol.h.col(step) = this->m_system->h(x_new_step, t_step);
1668
1669 // Check if the current step is the last one
1670 if (std::abs(t_step - t_mesh(last)) < SQRT_EPSILON) {break;}
1671
1672 // Update the previous step
1673 x_old_step = x_new_step;
1674
1675 // Propagate the derivative of the solution with respect to the states x
1676 if constexpr (Propagate) {Jx *= Jx_step;}
1677 }
1678 }
1679 return true;
1680
1681 #undef CMD
1682 }

1745 Integer safety_length{static_cast<Integer>(std::ceil(std::abs(t_mesh(last) - t_mesh(0))/(2.0*h_min)))};
1746 sol.resize(safety_length);
1747 } else {
1748 sol.resize(t_mesh.size());
1749 }
1750
1751 // Store initial conditions
1752 sol.t(0) = t_mesh(0);
1753 sol.x.col(0) = ics;
1754 sol.h.col(0) = this->m_system->h(ics, t_mesh(0));
1755
1756 // Reset the derivative propagation matrix
1757 if constexpr (Propagate) {Jx.setIdentity();}
1758
1759 // Instantiate temporary variables
1760 Integer step{0};
1761 VectorN x_old_step(ics), x_new_step(ics);
1762 MatrixJX Jx_step;
1763
1764 while (true) {
1765 // Integrate system
1766 this->template advance<Propagate>(x_old_step, t_step, h_step, x_new_step, h_new_step, Jx_step);
1767
1768 // Update the current step
1769 t_step += h_step;
1770
1771 // Saturate the suggested timestep
1772 if (this->m_adaptive && this->m_tableau.is_embedded) {
1773 h_step = std::max(std::min(h_new_step, h_max), h_min);
1774 }
1775
1776 SANDALS_ASSERT(step < sol.size(), CMD "safety length exceeded.");
1777
1778 // Update temporaries
1779 step += 1;
1780
1781 // Update outputs
1782 sol.t(step) = t_step;
1783 sol.x.col(step) = x_new_step;
1784 sol.h.col(step) = this->m_system->h(x_new_step, t_step);
1785
1786 // Check if the current step is the last one
1787 if (std::abs(t_step - t_mesh(last)) < SQRT_EPSILON) {break;}
1788 else if (t_step + h_step > t_mesh(last)) {h_step = t_mesh(last) - t_step;}
1789
1790 // Update the previous step
1791 x_old_step = x_new_step;
1792
1793 // Propagate the derivative of the solution with respect to the states x
1794 if constexpr (Propagate) {Jx *= Jx_step;}
1795 }
1796
1797 // Resize the output
1798 sol.conservative_resize(step);
1799
1800 return true;
1801
1802 #undef CMD
1803 }

1833
1834 // Check if there are any constraints
1835 x_projected = x;
1836 if (M > Integer(0)) {

1838 VectorM h;

1845
1846 /* Standard projection method
1847 [A] {x} = {b}
1848 / I Jh_x^T \ / dx \ / x_t - x_k \
1849 | | | | = | |
1850 \ Jh_x 0 / \ lambda / \ -h /
1851 */

1856

1859
1860 // Build the linear system
1861 A.template block<N, M>(0, N) = Jh_x.transpose();
1862 A.template block<M, N>(N, 0) = Jh_x;
1863 b.template head<N>() = x - x_projected;
1864 b.template tail<M>() = -h;

1867 this->m_lu.compute(A);
1868 SANDALS_ASSERT(this->m_lu.rank() == N+M, CMD "singular Jacobian detected.");

1870
1871 // Check if the step is too small

1873
1874 // Update the solution
1875 x_projected.noalias() += x_step(Eigen::seqN(0, N));
1876 }
1877 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
1878 return false;
1879 } else {

1969 Real estimate_order(std::vector<VectorX> const & t_mesh, VectorN const & ics, std::function<MatrixX(VectorX)> & sol) const
1970 {
1971 using Eigen::last;
1972
1973 #define CMD "Sandals::RungeKutta::estimate_order(...): "
1974
1975 SANDALS_ASSERT(t_mesh.size() > Integer(1), CMD "expected at least two time meshes.");
1976
1977 for (Integer i{0}; i < static_cast<Integer>(t_mesh.size()); ++i) {
1978
1979 // Check if the time mesh has the same initial and final time
1980 SANDALS_ASSERT((t_mesh[0](0) - t_mesh[i](0)) < SQRT_EPSILON,
1981 CMD "expected the same initial time.");
1982 SANDALS_ASSERT((t_mesh[0](last) - t_mesh[i](last)) < SQRT_EPSILON,
1983 CMD "expected the same final time.");
1984
1985 // Check if the mesh step is fixed all over the time mesh
1986 for (Integer j{1}; j < static_cast<Integer>(t_mesh[i].size()); ++j) {
1987 SANDALS_ASSERT((t_mesh[i](j) - t_mesh[i](j-1)) - (t_mesh[i](1) - t_mesh[i](0)) < SQRT_EPSILON,
1988 CMD "expected a fixed step.");
1989 }
1990 }
1991
1992 // Solve the system for each time scale
1993 Solution<Real, N, M> sol_num;
1994 MatrixX sol_ana;
1995 VectorX h_vec(t_mesh.size()), e_vec(t_mesh.size());
1996 for (Integer i{0}; i < static_cast<Integer>(t_mesh.size()); ++i) {
1997 SANDALS_ASSERT(this->solve(t_mesh[i], ics, sol_num), CMD "failed to solve the system for " <<
1998 "the" << i << "-th time mesh.");
1999 sol_ana = sol(sol_num.t);
2000 SANDALS_ASSERT(sol_ana.rows() == sol_num.x.rows(),
2001 CMD "expected the same number of states in analytical solution.");
2002 SANDALS_ASSERT(sol_ana.cols() == sol_num.x.cols(),
2003 CMD "expected the same number of steps in analytical solution.");
2004 h_vec(i) = std::abs(sol_num.t(1) - sol_num.t(0));
2005 e_vec(i) = (sol_ana - sol_num.x).array().abs().maxCoeff();
2006 }
2007
2008 // Compute the order of the method thorugh least squares
2009 VectorX A(h_vec.array().log());
2010 VectorX b(e_vec.array().log());
2011 return ((A.transpose() * A).ldlt().solve(A.transpose() * b))(0);
2012
2013 #undef CMD
2014 }
