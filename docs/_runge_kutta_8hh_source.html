679 Real estimate_step(VectorN const & x, VectorN const & x_e, Real const h_k) const
680 {
681 Real desired_error{this->m_absolute_tolerance + this->m_relative_tolerance *
682 std::max(x.array().abs().maxCoeff(), x_e.array().abs().maxCoeff())};
683 Real truncation_error{(x - x_e).array().abs().maxCoeff()};
684 return h_k * std::min(this->m_max_safety_factor, std::max(this->m_min_safety_factor,
685 this->m_safety_factor * std::pow(desired_error/truncation_error,
686 1.0/std::max(this->m_tableau.order, this->m_tableau.order_e))));
687 }

746 bool erk_explicit_step(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
747 Real & h_new, MatrixK & K) const
748 {
749 using Eigen::all;
750 using Eigen::seqN;
751
752 // Compute the K variables in the case of an explicit method and explicit system
753 VectorN x_node;
754 for (Integer i{0}; i < S; ++i) {
755 x_node = x_old + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
756 if (this->m_reverse) {
757 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f_reverse(x_node, t_old + h_old*this->m_tableau.c(i));
758 } else {
759 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f(x_node, t_old + h_old*this->m_tableau.c(i));
760 }
761 }
762 if (!K.allFinite()) {return false;}
763
764 // Perform the step and obtain the next state
765 x_new = x_old + K * this->m_tableau.b;
766
767 // Adapt next step
768 if (this->m_adaptive && this->m_tableau.is_embedded) {
769 VectorN x_emb(x_old + K * this->m_tableau.b_e);
770 h_new = this->estimate_step(x_new, x_emb, h_old);
771 }
772 return true;
773 }

788 bool erk_explicit_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
789 MatrixJX & Jx) const
790 {
791 #define CMD "Sandals::RungeKutta::erk_explicit_propagate(...): "
792
793 using Eigen::all;
794 using Eigen::seqN;
795
796 VectorN x_node;
797 MatrixN Jf_x;
798 std::array<MatrixN, S> dK_dx;
799 for (Integer i{0}; i < S; ++i) {
800 // Compute the node
801 x_node = x + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
802 (void)K;
803
804 // Compute the Jacobian of f with respect to x at the node
805 if (this->m_reverse) {
806 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x_reverse(x_node, t + h*this->m_tableau.c(i));
807 } else {
808 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x(x_node, t + h*this->m_tableau.c(i));
809 }
810
811 // Propagate the derivative of K with respect to x
812 if (i == 0) {
813 dK_dx[i] = h * Jf_x;
814 } else {
815 dK_dx[i].setIdentity();
816 for (Integer j{0}; j < i; ++j) {dK_dx[i] += this->m_tableau.A(i, j) * dK_dx[j];}
817 dK_dx[i] = h * Jf_x * dK_dx[i];
818 }
819
820 // Check for NaNs or Infs
821 if (!dK_dx[i].allFinite()) {return false;}
822 }
823
824 // Compute the derivative of x_new with respect to x
825 Jx.setIdentity();
826 for (Integer i{0}; i < S; ++i) {Jx += this->m_tableau.b(i) * dK_dx[i];}
827
828 #ifdef SANDALS_CHECK_JACOBIANS
829 // Function for the finite differences
830 auto fun = [this, t, h] (VectorN const & x_fd, VectorN & x_new_fd) -> bool {
831 Real h_fd;
832 MatrixK K_fd;
833 return this->erk_explicit_step(x_fd, t, h, x_new_fd, h_fd, K_fd);
834 };
835
836 // Compute the Jacobian with finite differences
837 MatrixJX Jx_fd;
838 if (Optimist::FiniteDifferences::Jacobian(x, fun, Jx_fd)) {
839 Real err{(Jx - Jx_fd).norm()};
840 SANDALS_ASSERT_WARNING(err < CBRT_EPSILON,
841 CMD "Jacobian propagation error = " << err << " > " << CBRT_EPSILON << ".");
842 }
843 #endif
844
845 return true;
846
847 #undef CMD
848 }

941 bool erk_implicit_step(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
942 Real & h_new, MatrixK & K) const
943 {

946
947 // Check if the solver converged

952 [this, s, &K, &x_old, t_old, h_old](VectorN const & K_jac, MatrixN & jac)

1136 void irk_jacobian(VectorN const & x, Real const t, Real const h, VectorK const & K, MatrixJ & jac) const
1137 {

1139

1141 jac.setZero();

1144 MatrixK K_mat{K.reshaped(N, S)};
1145 Real t_node;
1146 VectorN x_node, x_dot_node;
1147 MatrixN JF_x, JF_x_dot;
1148 auto idx = seqN(0, N), jdx = seqN(0, N);
1149 for (Integer i{0}; i < S; ++i) {
1150 t_node = t + h * this->m_tableau.c(i);
1151 x_node = x + K_mat * this->m_tableau.A.row(i).transpose();
1152

1154 x_dot_node = K_mat.col(i) / h;
1155 if (this->m_reverse) {
1156 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t_node);
1157 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t_node);

1160 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t_node);
1161 }

1164 idx = seqN(i*N, N);
1165 for (Integer j{0}; j < S; ++j) {
1166 jdx = seqN(j*N, N);

1168 jac(idx, jdx) = this->m_tableau.A(i, j) * JF_x + JF_x_dot / h;
1169 } else {
1170 jac(idx, jdx) = this->m_tableau.A(i, j) * JF_x;
1171 }

1233 bool irk_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,

1237
1238 using Eigen::all;
1239 using Eigen::seqN;

1241 // Propagate the derivative of K with respect to x
1242 VectorN x_node, x_dot_node;
1243 MatrixN JF_x, JF_x_dot;
1244 Eigen::Matrix<Real, N*S, N*S> A;
1245 Eigen::Matrix<Real, N*S, N> b;
1246 Eigen::FullPivLU<Eigen::Matrix<Real, N*S, N*S>> lu;
1247 for (Integer i{0}; i < S; ++i) {
1248 // Compute the node
1249 x_node = x + K * this->m_tableau.A.row(i).transpose();

1251
1252 // Compute the Jacobians of F with respect to x and x_dot at the node
1253 if (this->m_reverse) {
1254 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1255 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));

1257 JF_x = this->m_system->JF_x(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1258 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t + h*this->m_tableau.c(i));

1272 if (!dK_dx_mat.allFinite()) {return false;}

1274 // Compute the derivative of x_new with respect to x
1275 Jx.setIdentity();

1278 }

1600 template <bool Propagate = true>
1601 bool advance(VectorN const & x_old, Real const t_old, Real h_old, VectorN & x_new,
1602 Real & h_new, MatrixJX & Jx) const
1603 {
1604 #define CMD "Sandals::RungeKutta::advance(...): "
1605
1606 // Check step size
1607 SANDALS_ASSERT(h_old > 0.0,
1608 CMD "in " << this->m_tableau.name << ", inconsistent step size detected (h = " << h_old << ").");
1609
1610 // Reset the derivative propagation matrix
1611 if constexpr (Propagate) {Jx.setIdentity();}
1612
1613 // If the integration step failed, try again with substepping
1614 MatrixK K;
1615 if (!this->step(x_old, t_old, h_old, x_new, h_new, K)) {
1616
1617 // Store temporary variables
1618 VectorN x_tmp(x_old);
1619 Real t_tmp{t_old}, h_tmp{h_old / 2.0};
1620
1621 // Substepping logic
1622 Integer max_k{this->m_max_substeps * this->m_max_substeps}, k{2};
1623 Real h_new_tmp;
1624 while (k > 0) {
1625 // Calculate the next step with substepping logic
1626 if (this->step(x_tmp, t_tmp, h_tmp, x_new, h_new_tmp, K)) {
1627
1628 // Propagate the derivative of K with respect to x
1629 if constexpr (Propagate) {
1630 MatrixJX Jx_tmp;
1631 if (!this->propagate(x_tmp, t_tmp, h_tmp, K, Jx_tmp)) {
1632 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1633 ", Jacobian propagation failed.");
1634 return false;
1635 }
1636 Jx = Jx_tmp * Jx;
1637 }
1638
1639 // Accept the step
1640 h_tmp = h_new_tmp;
1641
1642 // If substepping is enabled, double the step size
1643 if (k > 0 && k < max_k) {
1644 k -= 1;
1645 // If the substepping index is even, double the step size
1646 if (k % 2 == 0) {
1647 h_tmp *= 2.0;
1648 if (this->m_verbose) {
1649 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1650 ", integration succedded disable one substepping layer.");
1651 }
1652 }
1653 }
1654
1655 // Check the infinity norm of the solution
1656 SANDALS_ASSERT(std::isfinite(x_tmp.maxCoeff()), CMD "in " << this->m_tableau.name <<
1657 " solver, at t = " << t_tmp << ", ||x||_inf = inf, computation interrupted.");
1658
1659 } else {
1660
1661 // If the substepping index is too high, abort the integration
1662 k += 2;
1663 SANDALS_ASSERT(k < max_k, CMD "in " << this->m_tableau.name << " solver, at t = " <<
1664 t_tmp << ", integration failed with h = " << h_tmp << ".");
1665 return false;
1666
1667 // Otherwise, try again with a smaller step
1668 if (this->m_verbose) {SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, " <<
1669 "at t = " << t_tmp << ", integration failed, adding substepping layer.");}
1670 h_tmp /= 2.0;
1671 continue;
1672 }
1673
1674 // Store independent variable (or time)
1675 t_tmp += h_tmp;
1676 }
1677
1678 // Store output states substepping solutions
1679 x_new = x_tmp;
1680 h_new = h_tmp;
1681
1682 } else {
1683 // Propagate the derivative of the solution with respect to the states x
1684 if constexpr (Propagate) {
1685 if (!this->propagate(x_old, t_old, h_old, K, Jx)) {
1686 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_old <<
1687 ", Jacobian propagation failed.");
1688 return false;
1689 }
1690 }
1691 }
1692
1693 // Project intermediate solution on the invariants
1694 if (this->m_projection) {
1695 VectorN x_projected;
1696 if (this->project(x_new, t_old + h_new, x_projected)) {
1697 x_new = x_projected;
1698 if constexpr (Propagate) {
1699 MatrixJX Jx_projected;
1700 if (!this->project_propagate(x_new, t_old + h_new, Jx_projected)) {
1701 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_old +
1702 h_new << ", projection Jacobian propagation failed.");
1703 return false;
1704 }
1705 Jx = Jx_projected * Jx;
1706 }
1707 } else {
1708 return false;
1709 }
1710 }
1711 return true;
1712
1713 #undef CMD
1714 }
