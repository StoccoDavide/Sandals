652 Real estimate_step(VectorN const & x, VectorN const & x_e, Real const h_k) const
653 {
654 Real desired_error{this->m_absolute_tolerance + this->m_relative_tolerance *
655 std::max(x.array().abs().maxCoeff(), x_e.array().abs().maxCoeff())};
656 Real truncation_error{(x - x_e).array().abs().maxCoeff()};
657 return h_k * std::min(this->m_max_safety_factor, std::max(this->m_min_safety_factor,
658 this->m_safety_factor * std::pow(desired_error/truncation_error,
659 1.0/std::max(this->m_tableau.order, this->m_tableau.order_e))));
660 }

719 bool erk_explicit_step(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
720 Real & h_new, MatrixK & K) const
721 {
722 using Eigen::all;
723 using Eigen::seqN;
724
725 // Compute the K variables in the case of an explicit method and explicit system
726 VectorN x_node;
727 for (Integer i{0}; i < S; ++i) {
728 x_node = x_old + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
729 if (!this->m_reverse) {
730 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f(x_node, t_old + h_old*this->m_tableau.c(i));
731 } else {
732 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f_reverse(x_node, t_old + h_old*this->m_tableau.c(i));
733 }
734 }
735 if (!K.allFinite()) {return false;}
736
737 // Perform the step and obtain the next state
738 x_new = x_old + K * this->m_tableau.b;
739
740 // Adapt next step
741 if (this->m_adaptive && this->m_tableau.is_embedded) {
742 VectorN x_emb = x_old + K * this->m_tableau.b_e;
743 h_new = this->estimate_step(x_new, x_emb, h_old);
744 }
745 return true;
746 }

761 bool erk_explicit_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
762 MatrixJX & Jx) const
763 {
764 #define CMD "Sandals::RungeKutta::erk_explicit_propagate(): "
765
766 using Eigen::all;
767 using Eigen::seqN;
768
769 // Propagate the derivative of K with respect to x
770 VectorN x_node;
771 MatrixN Jf_x;
772 std::array<MatrixN, S> dK_dx;
773 for (Integer i{0}; i < S; ++i) {
774 // Compute the node
775 x_node = x + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
776
777 // Compute the Jacobian of f with respect to x at the node
778 if (!this->m_reverse) {
779 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x(x_node, t + h*this->m_tableau.c(i));
780 } else {
781 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x_reverse(x_node, t + h*this->m_tableau.c(i));
782 }
783
784 // Propagate the derivative of K with respect to x
785 dK_dx[i] = h * Jf_x;
786 for (Integer j{0}; j < i; ++j) {
787 dK_dx[i] += h * Jf_x * dK_dx[j] * this->m_tableau.A(i, j);
788 }
789
790 // Check for NaNs or Infs
791 if (!dK_dx[i].allFinite()) {return false;}
792
793 }
794
795 // Compute the derivative of x_new with respect to x
796 Jx.setIdentity();
797 for (Integer i{0}; i < S; ++i) {Jx += h * dK_dx[i] * this->m_tableau.b(i);}
798
799 // Check the Jacobian with finite differences
800 #ifdef SANDALS_CHECK_JACOBIANS
801 MatrixJX Jx_fd;
802 auto fun = [this, &t, &h, &K](VectorN const & x_fd, VectorN & x_new_fd) -> bool {
803 MatrixK K_fd(K);
804 for (Integer k{0}; k < S; ++k) {
805 VectorN x_node_fd = x_fd + K_fd(Eigen::all, Eigen::seqN(0, k)) * this->m_tableau.A(k, Eigen::seqN(0, k)).transpose();
806 if (!this->m_reverse) {
807 K_fd.col(k) = h * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f(x_node_fd, t + h*this->m_tableau.c(k));
808 } else {
809 K_fd.col(k) = h * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f_reverse(x_node_fd, t + h*this->m_tableau.c(k));
810 }
811 }
812 x_new_fd = x_fd + K_fd * this->m_tableau.b;
813 return x_new_fd.allFinite();
814 };
815 bool fd_ok{Optimist::FiniteDifferences::Jacobian(x, fun, Jx_fd)};
816 if (fd_ok && Jx_fd.allFinite()) {
817 Real err{(Jx - Jx_fd).norm() / (1.0 + Jx_fd.norm())};
818 SANDALS_ASSERT_WARNING(err < EPSILON_LOW, CMD "Jacobian propagation error = " << err);
819 }
820 #endif
821
822 return true;
823
824 #undef CMD
825 }

1606 template <bool Propagate = true>
1607 bool advance(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
1608 Real & h_new, MatrixJX & Jx) const
1609 {
1610 #define CMD "Sandals::RungeKutta::advance(...): "
1611
1612 // Check step size
1613 SANDALS_ASSERT(h_old > Real(0.0), CMD "in " << this->m_tableau.name << " solver, h = "<<
1614 h_old << ", expected > 0.");
1615
1616 // Reset the derivative propagation matrix
1617 if constexpr (Propagate) {Jx.setIdentity();}
1618
1619 // If the integration step failed, try again with substepping
1620 MatrixK K;
1621 if (!this->step(x_old, t_old, h_old, x_new, h_new, K)) {
1622
1623 // Store temporary variables
1624 VectorN x_tmp(x_old);
1625 Real t_tmp{t_old}, h_tmp{h_old / Real(2.0)};
1626
1627 // Substepping logic
1628 Integer max_k{this->m_max_substeps * this->m_max_substeps}, k{2};
1629 Real h_new_tmp;
1630 while (k > 0) {
1631 // Calculate the next step with substepping logic
1632 if (this->step(x_tmp, t_tmp, h_tmp, x_new, h_new_tmp, K)) {
1633
1634 if constexpr (Propagate) {
1635 // Propagate the derivative of K with respect to x
1636 MatrixJX Jx_tmp;
1637 if (!this->propagate(x_tmp, t_tmp, h_tmp, K, Jx_tmp)) {
1638 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1639 ", Jacobian propagation failed, aborting.");
1640 return false;
1641 }
1642
1643 // Update the derivative propagation
1644 if constexpr (Propagate) {Jx *= Jx_tmp;}
1645 }
1646
1647 // Accept the step
1648 h_tmp = h_new_tmp;
1649
1650 // If substepping is enabled, double the step size
1651 if (k > 0 && k < max_k) {
1652 k -= 1;
1653 // If the substepping index is even, double the step size
1654 if (k % 2 == 0) {
1655 h_tmp = Real(2.0) * h_tmp;
1656 if (this->m_verbose) {
1657 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1658 ", integration succedded disable one substepping layer.");
1659 }
1660 }
1661 }
1662
1663 // Check the infinity norm of the solution

1708 }
1709 }

1731 {

1734 #define CMD "Sandals::RungeKutta::solve(...): "

1737 sol.resize(static_cast<Integer>(t_mesh.size()));

1743
1744 // Check initial conditions
1745 if constexpr (Propagate) {
1746 if (!Jx.allFinite()) {

1748 "contains NaNs or Infs, aborting.");

1750 }
1751 }
1752
1753 // Update the current step
1754 Integer step{0};

1761 // Integrate system

1765 // Update the current step
1766 t_step += h_step;
1767
1768 // Saturate the suggested timestep

1772 h_tmp_step = h_new_step; // Used to store the previous step and keep the integration pace
1773 h_step = t_mesh(step+1) - t_step;
1774 } else {
1775 h_step = h_new_step;
1776 }
1777
1778 // Propagate the derivative of the solution with respect to the states x
1779 if constexpr (Propagate) {Jx *= Jx_step;}
1780
1781 // Store solution if the step is a mesh point
1782 if (!this->m_adaptive || mesh_point_bool) {
1783
1784 // Update temporaries
1785 step += 1;
1786 h_step = h_tmp_step;
1787
1788 // Update outputs
1789 sol.t(step) = t_step;
1790 sol.x.col(step) = x_new_step;
1791 sol.h.col(step) = this->m_system->h(x_new_step, t_step);
1792
1793 // Check if the current step is the last one
1794 if (std::abs(t_step - t_mesh(last)) < SQRT_EPSILON) {break;}
1795
1796 // Update the previous step
1797 x_old_step = x_new_step;
1798
1799 }
1800
1801 }

1953 bool project(VectorN const & x, Real const t, VectorN & x_projected) const
1954 {
1955 #define CMD "Sandals::RungeKutta::project(...): "
1956
1957 // Check if there are any constraints
1958 x_projected = x;
1959 if (M > Integer(0)) {
1960
1961 VectorM h;
1962 MatrixM Jh_x;
1963 VectorP b, x_step;
1964 MatrixP A;
1965 A.setZero();
1966 A.template block<N, N>(0, 0) = MatrixN::Identity();
1967 for (Integer k{0}; k < this->m_max_projection_iterations; ++k) {
1968
1969 /* Standard projection method
1970 [A] {x} = {b}
1971 / I Jh_x^T \ / dx \ / x_t - x_k \
1972 | | | | = | |
1973 \ Jh_x 0 / \ lambda / \ -h /
1974 */
1975
1976 // Evaluate the invariants vector and its Jacobian
1977 h = this->m_system->h(x_projected, t);
1978 Jh_x = this->m_system->Jh_x(x_projected, t);
1979
1980 // Check if the solution is found
1981 if (h.norm() < this->m_projection_tolerance) {return true;}
1982
1983 // Build the linear system
1984 A.template block<N, M>(0, N) = Jh_x.transpose();
1985 A.template block<M, N>(N, 0) = Jh_x;
1986 b.template head<N>() = x - x_projected;
1987 b.template tail<M>() = -h;
1988
1989 // Compute the solution of the linear system
1990 this->m_lu.compute(A);
1991 SANDALS_ASSERT(this->m_lu.rank() == N+M, CMD "singular Jacobian detected.");
1992 x_step = this->m_lu.solve(b);
1993
1994 // Check if the step is too small
1995 if (x_step.norm() < this->m_projection_tolerance * this->m_projection_tolerance) {return false;}
1996
1997 // Update the solution
1998 x_projected.noalias() += x_step(Eigen::seqN(0, N));
1999 }
2000 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
2001 return false;
2002 } else {
2003 return true;
2004 }

2006 #undef CMD
