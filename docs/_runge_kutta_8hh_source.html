653 Real estimate_step(VectorN const & x, VectorN const & x_e, Real const h_k) const
654 {
655 Real desired_error{this->m_absolute_tolerance + this->m_relative_tolerance *
656 std::max(x.array().abs().maxCoeff(), x_e.array().abs().maxCoeff())};
657 Real truncation_error{(x - x_e).array().abs().maxCoeff()};
658 return h_k * std::min(this->m_max_safety_factor, std::max(this->m_min_safety_factor,
659 this->m_safety_factor * std::pow(desired_error/truncation_error,
660 1.0/std::max(this->m_tableau.order, this->m_tableau.order_e))));
661 }

720 bool erk_explicit_step(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
721 Real & h_new, MatrixK & K) const
722 {
723 using Eigen::all;
724 using Eigen::seqN;
725
726 // Compute the K variables in the case of an explicit method and explicit system
727 VectorN x_node;
728 for (Integer i{0}; i < S; ++i) {
729 x_node = x_old + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
730 if (!this->m_reverse) {
731 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f(x_node, t_old + h_old*this->m_tableau.c(i));
732 } else {
733 K.col(i) = h_old * static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->f_reverse(x_node, t_old + h_old*this->m_tableau.c(i));
734 }
735 }
736 if (!K.allFinite()) {return false;}
737
738 // Perform the step and obtain the next state
739 x_new = x_old + K * this->m_tableau.b;
740
741 // Adapt next step
742 if (this->m_adaptive && this->m_tableau.is_embedded) {
743 VectorN x_emb = x_old + K * this->m_tableau.b_e;
744 h_new = this->estimate_step(x_new, x_emb, h_old);
745 }
746 return true;
747 }

762 bool erk_explicit_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
763 MatrixJX & Jx) const
764 {
765 #define CMD "Sandals::RungeKutta::erk_explicit_propagate(): "
766
767 using Eigen::all;
768 using Eigen::seqN;
769
770 VectorN x_node;
771 MatrixN Jf_x;
772 std::array<MatrixN, S> dK_dx;
773 for (Integer i{0}; i < S; ++i) {
774 // Compute the node
775 x_node = x + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
776 (void)K;
777
778 // Compute the Jacobian of f with respect to x at the node
779 if (!this->m_reverse) {
780 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x(x_node, t + h*this->m_tableau.c(i));
781 } else {
782 Jf_x = static_cast<Explicit<Real, N, M> const *>(this->m_system.get())->Jf_x_reverse(x_node, t + h*this->m_tableau.c(i));
783 }
784
785 // Propagate the derivative of K with respect to x
786 if (i == 0) {
787 dK_dx[i] = h * Jf_x;
788 } else {
789 dK_dx[i].setIdentity();
790 for (Integer j{0}; j < i; ++j) {dK_dx[i] += this->m_tableau.A(i, j) * dK_dx[j];}
791 dK_dx[i] = h * Jf_x * dK_dx[i];
792 }
793
794 // Check for NaNs or Infs
795 if (!dK_dx[i].allFinite()) {return false;}
796 }
797
798 // Compute the derivative of x_new with respect to x
799 Jx.setIdentity();
800 for (Integer i{0}; i < S; ++i) {Jx += this->m_tableau.b(i) * dK_dx[i];}
801
802 #ifdef SANDALS_CHECK_JACOBIANS
803 // Function for the finite differences
804 auto fun = [this, t, h] (VectorN const & x_fd, VectorN & x_new_fd) -> bool {
805 Real h_fd;
806 MatrixK K_fd;
807 return this->erk_explicit_step(x_fd, t, h, x_new_fd, h_fd, K_fd);
808 };
809
810 // Compute the Jacobian with finite differences
811 MatrixJX Jx_fd;
812 if (Optimist::FiniteDifferences::Jacobian(x, fun, Jx_fd)) {
813 Real err{(Jx - Jx_fd).norm()};
814 SANDALS_ASSERT_WARNING(err < CBRT_EPSILON,
815 CMD "Jacobian propagation error = " << err << " > " << CBRT_EPSILON << ".");
816 }
817 #endif
818
819 return true;
820
821 #undef CMD
822 }

916 Real & h_new, MatrixK & K) const
917 {

920
921 // Check if the solver converged

925 {K.col(s) = K_fun; this->erk_implicit_function(s, x_old, t_old, h_old, K, fun);},
926 [this, s, &K, &x_old, t_old, h_old](VectorN const & K_jac, MatrixN & jac)
927 {K.col(s) = K_jac; this->erk_implicit_jacobian(s, x_old, t_old, h_old, K, jac);},
928 K_ini, K_sol)) {

932 }
933 }

959 bool erk_implicit_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
960 MatrixJX & Jx) const
961 {
962 #define CMD "Sandals::RungeKutta::erk_implicit_propagate(): "
963
964 using Eigen::all;
965 using Eigen::seqN;
966
967 VectorN x_node, x_dot_node;
968 MatrixN JF_x, JF_x_dot, A, b;
969 std::array<MatrixN, S> dK_dx;
970 Eigen::FullPivLU<MatrixN> lu;
971 for (Integer i{0}; i < S; ++i) {
972 // Compute the node
973 x_node = x + K(all, seqN(0, i)) * this->m_tableau.A(i, seqN(0, i)).transpose();
974 x_dot_node = K.col(i) / h;
975

977 if (!this->m_reverse) {

987 b.setIdentity();

994

996 if (!dK_dx[i].allFinite()) {return false;}
997 }

1001 for (Integer i{0}; i < S; ++i) {Jx += this->m_tableau.b(i) * dK_dx[i];}

1110 void irk_jacobian(VectorN const & x, Real const t, Real const h, VectorK const & K, MatrixJ & jac) const
1111 {

1113
1114 // Reset the Jacobian matrix
1115 jac.setZero();

1117 // Loop through each equation of the system
1118 MatrixK K_mat{K.reshaped(N, S)};
1119 Real t_node;
1120 VectorN x_node, x_dot_node;
1121 MatrixN JF_x, JF_x_dot;
1122 auto idx = seqN(0, N), jdx = seqN(0, N);
1123 for (Integer i{0}; i < S; ++i) {
1124 t_node = t + h * this->m_tableau.c(i);
1125 x_node = x + K_mat * this->m_tableau.A.row(i).transpose();
1126

1128 x_dot_node = K_mat.col(i) / h;
1129 if (!this->m_reverse) {

1131 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t_node);

1134 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t_node);
1135 }
1136

1138 idx = seqN(i*N, N);
1139 for (Integer j{0}; j < S; ++j) {
1140 jdx = seqN(j*N, N);
1141 if (i == j) {
1142 jac(idx, jdx) = this->m_tableau.A(i, j) * JF_x + JF_x_dot / h;
1143 } else {
1144 jac(idx, jdx) = this->m_tableau.A(i, j) * JF_x;
1145 }

1207 bool irk_propagate(VectorN const & x, Real const t, Real const h, MatrixK const & K,
1208 MatrixJX & Jx) const
1209 {
1210 #define CMD "Sandals::RungeKutta::irk_propagate(): "
1211
1212 using Eigen::all;
1213 using Eigen::seqN;
1214
1215 // Propagate the derivative of K with respect to x
1216 VectorN x_node, x_dot_node;
1217 MatrixN JF_x, JF_x_dot;
1218 Eigen::Matrix<Real, N*S, N*S> A;
1219 Eigen::Matrix<Real, N*S, N> b;
1220 Eigen::FullPivLU<Eigen::Matrix<Real, N*S, N*S>> lu;
1221 for (Integer i{0}; i < S; ++i) {
1222 // Compute the node
1223 x_node = x + K * this->m_tableau.A.row(i).transpose();
1224 x_dot_node = K.col(i) / h;
1225
1226 // Compute the Jacobians of F with respect to x and x_dot at the node
1227 if (!this->m_reverse) {
1228 JF_x = this->m_system->JF_x(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1229 JF_x_dot = this->m_system->JF_x_dot(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1230 } else {
1231 JF_x = this->m_system->JF_x_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1232 JF_x_dot = this->m_system->JF_x_dot_reverse(x_node, x_dot_node, t + h*this->m_tableau.c(i));
1233 }
1234
1235 // Fill the big linear system
1236 for (Integer j{0}; j < S; ++j) {
1237 A.template block<N, N>(i*N, j*N) = (h * this->m_tableau.A(i, j)) * JF_x;
1238 }
1239 A.template block<N, N>(i*N, i*N) += JF_x_dot;
1240 b.template block<N, N>(i*N, 0) = -h * JF_x;
1241 }
1242
1243 // Solve the big linear system
1244 lu.compute(A);
1245 Eigen::Matrix<Real, N*S, N> dK_dx_mat(lu.solve(b));
1246 if (!dK_dx_mat.allFinite()) {return false;}
1247
1248 // Compute the derivative of x_new with respect to x
1249 Jx.setIdentity();
1250 for (Integer i{0}; i < S; ++i) {
1251 Jx += this->m_tableau.b(i) * dK_dx_mat.template block<N, N>(i*N, 0);
1252 }

1254 #ifdef SANDALS_CHECK_JACOBIANS
1255 // Function for the finite differences
1256 auto fun = [this, t, h] (VectorN const & x_fd, VectorN & x_new_fd) -> bool {
1257 Real h_fd;

1259 return this->irk_step(x_fd, t, h, x_new_fd, h_fd, K_fd);
1260 };
1261
1262 // Compute the Jacobian with finite differences
1263 MatrixJX Jx_fd;
1264 if (Optimist::FiniteDifferences::Jacobian(x, fun, Jx_fd)) {
1265 Real err{(Jx - Jx_fd).norm()};
1266 SANDALS_ASSERT_WARNING(err < CBRT_EPSILON,
1267 CMD "Jacobian propagation at time t = " << t << ", error = " << err << " > " << CBRT_EPSILON << ".");
1268 }
1269 #endif
1270
1271 return true;

1273 #undef CMD

1574 template <bool Propagate = true>
1575 bool advance(VectorN const & x_old, Real const t_old, Real const h_old, VectorN & x_new,
1576 Real & h_new, MatrixJX & Jx) const
1577 {
1578 #define CMD "Sandals::RungeKutta::advance(...): "
1579
1580 // Check step size
1581 SANDALS_ASSERT(h_old > Real(0.0), CMD "in " << this->m_tableau.name << " solver, h = "<<
1582 h_old << ", expected > 0.");
1583
1584 // Reset the derivative propagation matrix
1585 if constexpr (Propagate) {Jx.setIdentity();}
1586
1587 // If the integration step failed, try again with substepping
1588 MatrixK K;
1589 if (!this->step(x_old, t_old, h_old, x_new, h_new, K)) {
1590
1591 // Store temporary variables
1592 VectorN x_tmp(x_old);
1593 Real t_tmp{t_old}, h_tmp{h_old / Real(2.0)};
1594
1595 // Substepping logic
1596 Integer max_k{this->m_max_substeps * this->m_max_substeps}, k{2};
1597 Real h_new_tmp;
1598 while (k > 0) {
1599 // Calculate the next step with substepping logic
1600 if (this->step(x_tmp, t_tmp, h_tmp, x_new, h_new_tmp, K)) {
1601
1602 if constexpr (Propagate) {
1603 // Propagate the derivative of K with respect to x
1604 MatrixJX Jx_tmp;
1605 if (!this->propagate(x_tmp, t_tmp, h_tmp, K, Jx_tmp)) {
1606 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1607 ", Jacobian propagation failed, aborting.");
1608 return false;
1609 }
1610
1611 // Update the derivative propagation
1612 if constexpr (Propagate) {Jx = Jx_tmp * Jx;}
1613 }
1614
1615 // Accept the step
1616 h_tmp = h_new_tmp;
1617
1618 // If substepping is enabled, double the step size
1619 if (k > 0 && k < max_k) {
1620 k -= 1;
1621 // If the substepping index is even, double the step size
1622 if (k % 2 == 0) {
1623 h_tmp = Real(2.0) * h_tmp;
1624 if (this->m_verbose) {
1625 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_tmp <<
1626 ", integration succedded disable one substepping layer.");
1627 }
1628 }
1629 }
1630
1631 // Check the infinity norm of the solution
1632 SANDALS_ASSERT(std::isfinite(x_tmp.maxCoeff()), CMD "in " << this->m_tableau.name <<
1633 " solver, at t = " << t_tmp << ", ||x||_inf = inf, computation interrupted.");
1634
1635 } else {
1636
1637 // If the substepping index is too high, abort the integration
1638 k += 2;
1639 SANDALS_ASSERT(k < max_k, CMD "in " << this->m_tableau.name << " solver, at t = " <<
1640 t_tmp << ", integration failed with h = " << h_tmp << ", aborting.");
1641 return false;
1642
1643 // Otherwise, try again with a smaller step
1644 if (this->m_verbose) {SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, " <<
1645 "at t = " << t_tmp << ", integration failed, adding substepping layer.");}
1646 h_tmp /= Real(2.0);
1647 continue;
1648 }
1649
1650 // Store independent variable (or time)
1651 t_tmp += h_tmp;
1652 }
1653
1654 // Store output states substepping solutions
1655 x_new = x_tmp;
1656 h_new = h_tmp;
1657
1658 } else {
1659 // Propagate the derivative of the solution with respect to the states x
1660 if constexpr (Propagate) {
1661 if (!this->propagate(x_old, t_old, h_old, K, Jx)) {
1662 SANDALS_WARNING(CMD "in " << this->m_tableau.name << " solver, at t = " << t_old <<
1663 ", Jacobian propagation failed, aborting.");
1664 return false;
1665 }
1666 }
1667 }
1668
1669 // Project intermediate solution on the invariants
1670 if (this->m_projection) {
1671 VectorN x_projected;
1672 if (this->project(x_new, t_old + h_new, x_projected)) {
1673 x_new = x_projected;
1674 } else {
1675 return false;
1676 }
1677 }
1678 return true;
1679
1680 #undef CMD
1681 }

1697 template <bool Propagate = true>
1698 bool solve(VectorX const & t_mesh, VectorN const & ics, Solution<Real, N, M> & sol, MatrixJX & Jx) const
1699 {
1700 using Eigen::last;
1701
1702 #define CMD "Sandals::RungeKutta::solve(...): "
1703
1704 // Instantiate output
1705 sol.resize(static_cast<Integer>(t_mesh.size()));
1706
1707 // Store initial conditions
1708 sol.t(0) = t_mesh(0);
1709 sol.x.col(0) = ics;
1710 sol.h.col(0) = this->m_system->h(ics, t_mesh(0));
1711
1712 // Check initial conditions
1713 if constexpr (Propagate) {
1714 if (!Jx.allFinite()) {
1715 SANDALS_ERROR(CMD "in " << this->m_tableau.name << " solver, initial Jacobian " <<
1716 "contains NaNs or Infs, aborting.");

1721 // Update the current step

1922 {
1923 #define CMD "Sandals::RungeKutta::project(...): "
1924
1925 // Check if there are any constraints
1926 x_projected = x;
1927 if (M > Integer(0)) {
1928
1929 VectorM h;
1930 MatrixM Jh_x;
1931 VectorP b, x_step;
1932 MatrixP A;
1933 A.setZero();
1934 A.template block<N, N>(0, 0) = MatrixN::Identity();
1935 for (Integer k{0}; k < this->m_max_projection_iterations; ++k) {
1936
1937 /* Standard projection method
1938 [A] {x} = {b}
1939 / I Jh_x^T \ / dx \ / x_t - x_k \
1940 | | | | = | |
1941 \ Jh_x 0 / \ lambda / \ -h /
1942 */
1943
1944 // Evaluate the invariants vector and its Jacobian
1945 h = this->m_system->h(x_projected, t);
1946 Jh_x = this->m_system->Jh_x(x_projected, t);
1947
1948 // Check if the solution is found
1949 if (h.norm() < this->m_projection_tolerance) {return true;}
1950
1951 // Build the linear system
1952 A.template block<N, M>(0, N) = Jh_x.transpose();
1953 A.template block<M, N>(N, 0) = Jh_x;
1954 b.template head<N>() = x - x_projected;
1955 b.template tail<M>() = -h;
1956
1957 // Compute the solution of the linear system
1958 this->m_lu.compute(A);
1959 SANDALS_ASSERT(this->m_lu.rank() == N+M, CMD "singular Jacobian detected.");
1960 x_step = this->m_lu.solve(b);
1961
1962 // Check if the step is too small
1963 if (x_step.norm() < this->m_projection_tolerance * this->m_projection_tolerance) {return false;}
1964
1965 // Update the solution
1966 x_projected.noalias() += x_step(Eigen::seqN(0, N));
1967 }
1968 if (this->m_verbose) {SANDALS_WARNING(CMD "maximum number of iterations reached.");}
1969 return false;
1970 } else {
1971 return true;
1972 }

2081 }
2082
2083 // Solve the system for each time scale
2084 Solution<Real, N, M> sol_num;
2085 MatrixX sol_ana;
2086 VectorX h_vec(t_mesh.size()), e_vec(t_mesh.size());
2087 for (Integer i{0}; i < static_cast<Integer>(t_mesh.size()); ++i) {

2092 CMD "expected the same number of states in analytical solution.");
2093 SANDALS_ASSERT(sol_ana.cols() == sol_num.x.cols(),
2094 CMD "expected the same number of steps in analytical solution.");

2096 e_vec(i) = (sol_ana - sol_num.x).array().abs().maxCoeff();

2099 // Compute the order of the method thorugh least squares
2100 VectorX A(h_vec.array().log());
2101 VectorX b(e_vec.array().log());
